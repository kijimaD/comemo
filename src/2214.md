+func TestPowx(t *testing.T) {
+	p := 1;
+	for i := 0; i < len(powx); i++ {
+		if powx[i] != byte(p) {
+			t.Errorf("powx[%d] = %#x, want %#x", i, powx[i], p);
+		}
+		p <<= 1;
+		if p & 0x100 != 0 {
+			p ^= poly;
+		}
+	}

+// Multiply b and c as GF(2) polynomials modulo poly
+func mul(b, c uint32) uint32 {
+	i := b;
+	j := c;
+	s := uint32(0);
+	for k := uint32(1); k < 0x100 && j != 0; k <<= 1 {
+		// Invariant: k == 1<<n, i == b * xⁿ
+
+		if j & k != 0 {
+			// s += i in GF(2); xor in binary
+			s ^= i;
+			j ^= k;	// turn off bit to end loop early
+		}
+
+		// i *= x in GF(2) modulo the polynomial
+		i <<= 1;
+		if i & 0x100 != 0 {
+			i ^= poly;
+		}
+	}
+	return s;
+}

+// Test all mul inputs against bit-by-bit n² algorithm.
+func TestMul(t *testing.T) {
+	for i := uint32(0); i < 256; i++ {
+		for j := uint32(0); j < 256; j++ {
+			// Multiply i, j bit by bit.
+			s := uint8(0);
+			for k := uint(0); k < 8; k++ {
+				for l := uint(0); l < 8; l++ {
+					if i & (1<<k) != 0 && j & (1<<l) != 0 {
+						s ^= powx[k+l];
+					}
+				}
+			}
+			if x := mul(i, j); x != uint32(s) {
+				t.Fatalf("mul(%#x, %#x) = %#x, want %#x", i, j, x, s);
+			}
+		}
+	}

+// Encrypt one block from src into dst, using the expanded key xk.
+func encryptBlock(xk, src, dst []uint32) {
+	var s0, s1, s2, s3, t0, t1, t2, t3 uint32;
+
+	// First round just XORs input with key.
+	s0 = src[0] ^ xk[0];
+	s1 = src[1] ^ xk[1];
+	s2 = src[2] ^ xk[2];
+	s3 = src[3] ^ xk[3];
+
+	// Middle rounds shuffle using tables.
+	// Number of rounds is set by length of expanded key.
+	nr := len(xk)/4 - 2;	// - 2: one above, one more below
+	k := 4;\
+	for r := 0; r < nr; r++ {\
+		t0 = xk[k+0] ^ te[0][s0>>24] ^ te[1][s1>>16 & 0xff] ^ te[2][s2>>8 & 0xff] ^ te[3][s3 & 0xff];\
+		t1 = xk[k+1] ^ te[0][s1>>24] ^ te[1][s2>>16 & 0xff] ^ te[2][s3>>8 & 0xff] ^ te[3][s0 & 0xff];\
+		t2 = xk[k+2] ^ te[0][s2>>24] ^ te[1][s3>>16 & 0xff] ^ te[2][s0>>8 & 0xff] ^ te[3][s1 & 0xff];\
+		t3 = xk[k+3] ^ te[0][s3>>24] ^ te[1][s0>>16 & 0xff] ^ te[2][s1>>8 & 0xff] ^ te[3][s2 & 0xff];\
+		k += 4;\
+		s0, s1, s2, s3 = t0, t1, t2, t3;\
+	}\
+
+	// Last round uses s-box directly and XORs to produce output.
+	s0 = uint32(sbox0[t0>>24])<<24 | uint32(sbox0[t1>>16 & 0xff])<<16 | uint32(sbox0[t2>>8 & 0xff])<<8 | uint32(sbox0[t3 & 0xff]);\
+	s1 = uint32(sbox0[t1>>24])<<24 | uint32(sbox0[t2>>16 & 0xff])<<16 | uint32(sbox0[t3>>8 & 0xff])<<8 | uint32(sbox0[t0 & 0xff]);\
+	s2 = uint32(sbox0[t2>>24])<<24 | uint32(sbox0[t3>>16 & 0xff])<<16 | uint32(sbox0[t0>>8 & 0xff])<<8 | uint32(sbox0[t1 & 0xff]);\
+	s3 = uint32(sbox0[t3>>24])<<24 | uint32(sbox0[t0>>16 & 0xff])<<16 | uint32(sbox0[t1>>8 & 0xff])<<8 | uint32(sbox0[t2 & 0xff]);\
+
+	dst[0] = s0 ^ xk[k+0];\
+	dst[1] = s1 ^ xk[k+1];\
+	dst[2] = s2 ^ xk[k+2];\
+	dst[3] = s3 ^ xk[k+3];
+}\

+// Decrypt one block from src into dst, using the expanded key xk.
+func decryptBlock(xk, src, dst []uint32) {
+	var s0, s1, s2, s3, t0, t1, t2, t3 uint32;
+
+	// First round just XORs input with key.
+	s0 = src[0] ^ xk[0];
+	s1 = src[1] ^ xk[1];
+	s2 = src[2] ^ xk[2];
+	s3 = src[3] ^ xk[3];
+
+	// Middle rounds shuffle using tables.
+	// Number of rounds is set by length of expanded key.
+	nr := len(xk)/4 - 2;	// - 2: one above, one more below
+	k := 4;\
+	for r := 0; r < nr; r++ {\
+		t0 = xk[k+0] ^ td[0][s0>>24] ^ td[1][s3>>16 & 0xff] ^ td[2][s2>>8 & 0xff] ^ td[3][s1 & 0xff];\
+		t1 = xk[k+1] ^ td[0][s1>>24] ^ td[1][s0>>16 & 0xff] ^ td[2][s3>>8 & 0xff] ^ td[3][s2 & 0xff];\
+		t2 = xk[k+2] ^ td[0][s2>>24] ^ td[1][s1>>16 & 0xff] ^ td[2][s0>>8 & 0xff] ^ td[3][s3 & 0xff];\
+		t3 = xk[k+3] ^ td[0][s3>>24] ^ td[1][s2>>16 & 0xff] ^ td[2][s1>>8 & 0xff] ^ td[3][s0 & 0xff];\
+		k += 4;\
+		s0, s1, s2, s3 = t0, t1, t2, t3;\
+	}\
+
+	// Last round uses s-box directly and XORs to produce output.
+	s0 = uint32(sbox1[t0>>24])<<24 | uint32(sbox1[t3>>16 & 0xff])<<16 | uint32(sbox1[t2>>8 & 0xff])<<8 | uint32(sbox1[t1 & 0xff]);\
+	s1 = uint32(sbox1[t1>>24])<<24 | uint32(sbox1[t0>>16 & 0xff])<<16 | uint32(sbox1[t3>>8 & 0xff])<<8 | uint32(sbox1[t2 & 0xff]);\
+	s2 = uint32(sbox1[t2>>24])<<24 | uint32(sbox1[t1>>16 & 0xff])<<16 | uint32(sbox1[t0>>8 & 0xff])<<8 | uint32(sbox1[t3 & 0xff]);\
+	s3 = uint32(sbox1[t3>>24])<<24 | uint32(sbox1[t2>>16 & 0xff])<<16 | uint32(sbox1[t1>>8 & 0xff])<<8 | uint32(sbox1[t0 & 0xff]);\
+
+	dst[0] = s0 ^ xk[k+0];\
+	dst[1] = s1 ^ xk[k+1];\
+	dst[2] = s2 ^ xk[k+2];\
+	dst[3] = s3 ^ xk[k+3];
+}\
+
+// Apply sbox0 to each byte in w.\
+func subw(w uint32) uint32 {\
+	return\
+		uint32(sbox0[w>>24])<<24 |\
+		uint32(sbox0[w>>16 & 0xff])<<16 |\
+		uint32(sbox0[w>>8 & 0xff])<<8 |\
+		uint32(sbox0[w & 0xff]);\
+}\
+
+// Rotate\
+func rotw(w uint32) uint32 {\
+	return w<<8 | w>>24;\
+}\
+
+// Key expansion algorithm.  See FIPS-197, Figure 11.\
+// Their rcon[i] is our powx[i-1] << 24.\
+func expandKey(key, enc, dec []uint32) {\
+	// Encryption key setup.\
+	var i int;\
+	nk := len(key);\
+	for i = 0; i < nk; i++ {\
+		enc[i] = key[i];\
+	}\
+	for ; i < len(enc); i++ {\
+		t := enc[i-1];\
+		if i % nk == 0 {\
+			t = subw(rotw(t)) ^ (uint32(powx[i/nk - 1]) << 24);\
+		} else if nk > 6 && i % nk == 4 {\
+			t = subw(t);\
+		}\
+		enc[i] = enc[i-nk] ^ t;\
+	}\
+
+	// Derive decryption key from encryption key.\
+	// Reverse the 4-word round key sets from enc to produce dec.\
+	// All sets but the first and last get the MixColumn transform applied.\
+	if dec == nil {\
+		return;\
+	}\
+	n := len(enc);\
+	for i := 0; i < n; i += 4 {\
+		ei := n - i - 4;\
+		for j := 0; j < 4; j++ {\
+			x := enc[ei+j];\
+			if i > 0 && i+4 < n {\
+				x = td[0][sbox0[x>>24]] ^ td[1][sbox0[x>>16 & 0xff]] ^ td[2][sbox0[x>>8 & 0xff]] ^ td[3][sbox0[x & 0xff]];\
+			}\
+			dec[i+j] = x;\
+		}\
+	}\
+}\

## 元コミット内容

AESの鍵設定とブロック暗号化・復号化の実装。
AESモードのラッパーはまだ実装されていないため、現時点では公開インターフェースは提供されていません。

## 変更の背景

このコミットは、Go言語の標準ライブラリに暗号化機能の基盤を導入する初期段階の一部です。AES（Advanced Encryption Standard）は、現代の暗号化において最も広く利用されている対称鍵暗号アルゴリズムの一つであり、その実装は多くのアプリケーションにとって不可欠です。このコミットの目的は、AESのコアとなるブロック暗号化・復号化ロジックと鍵スケジューリング（鍵拡張）をGoで利用できるようにすることでした。

当時のGo言語はまだ開発初期段階であり、標準ライブラリの拡充が積極的に行われていました。暗号化機能はセキュリティを重視するシステムにとって必須であり、AESの実装はその中でも優先度の高い課題でした。このコミットは、将来的に様々な暗号化モード（CBC、GCMなど）を実装するための低レベルなプリミティブを提供することを意図しています。コミットメッセージにある「AES mode wrappers not implemented, so no public interface yet.」という記述は、この時点ではまだエンドユーザーが直接利用できる高レベルなAPIは提供されていないことを示唆しています。

## 前提知識の解説

AESの技術的詳細を理解するためには、いくつかの暗号学および数学の基礎知識が必要です。

### 1. 有限体 (Galois Field, GF)

AESは、有限体GF(2^8)上での演算に基づいています。有限体とは、要素の数が有限である体（四則演算が定義され、特定の性質を満たす代数構造）のことです。GF(2^8)は、256個の要素を持つ体であり、各要素は8ビットのバイナリデータ（バイト）として表現されます。

*   **GF(2)上の多項式**: GF(2)は要素が0と1のみの体です。GF(2)上の多項式とは、係数が0または1である多項式のことです。例えば、x^7 + x^5 + x + 1 など。
*   **加算**: GF(2)上の多項式の加算は、対応する係数ごとの排他的論理和（XOR）です。
*   **乗算**: GF(2)上の多項式の乗算は、通常の多項式乗算を行った後、既約多項式（AESでは `x^8 + x^4 + x^3 + x + 1`）で割った余りを取ります。この除算により、結果が常に8ビットに収まります。

### 2. AESの基本構造

AESは、ブロック暗号であり、128ビット（16バイト）の固定長ブロックを暗号化・復号化します。鍵長は128ビット、192ビット、256ビットのいずれかを選択できます。鍵長によってラウンド数（変換処理の繰り返し回数）が異なります。

*   **128ビット鍵**: 10ラウンド
*   **192ビット鍵**: 12ラウンド
*   **256ビット鍵**: 14ラウンド

各ラウンドは、以下の4つの主要な変換ステップで構成されます（最終ラウンドを除く）。

*   **SubBytes (バイト置換)**: 各バイトを非線形なS-Box（Substitution Box）を使って別のバイトに置換します。これはGF(2^8)上での逆元計算とアフィン変換の組み合わせによって実現されます。暗号の非線形性を導入し、線形攻撃を防ぐ役割があります。
*   **ShiftRows (行シフト)**: 状態行列（4x4のバイト行列）の各行を異なる量だけ左に巡回シフトします。これにより、列間の拡散（diffusion）が促進されます。
    *   1行目: 0バイトシフト
    *   2行目: 1バイトシフト
    *   3行目: 2バイトシフト
    *   4行目: 3バイトシフト
*   **MixColumns (列混合)**: 状態行列の各列を、GF(2^8)上での多項式乗算によって混合します。これにより、行間の拡散が促進され、各バイトが他の多くのバイトに影響を与えるようになります。
*   **AddRoundKey (ラウンド鍵加算)**: 現在のラウンド鍵と状態行列をビットごとのXORで結合します。ラウンド鍵は、元の秘密鍵から鍵拡張（Key Expansion）アルゴリズムによって生成されます。

### 3. 鍵拡張 (Key Expansion)

AESは、元の秘密鍵から各ラウンドで使用するラウンド鍵を生成する鍵拡張アルゴリズムを持っています。このプロセスは、元の鍵を基に、SubWord、RotWord、Rcon（ラウンド定数）などの操作を組み合わせて行われます。鍵拡張は、暗号化と復号化の両方で必要となります。

### 4. S-Boxと逆S-Box

S-Boxは、SubBytes変換で使用される256エントリのルックアップテーブルです。各バイトを入力として受け取り、別のバイトを出力します。逆S-Boxは、復号化の際に使用され、S-Boxの逆変換を行います。これらのS-Boxは、GF(2^8)上での逆元計算とアフィン変換によって数学的に導出されます。

### 5. ルックアップテーブル (Lookup Tables)

AESのMixColumns変換は、GF(2^8)上での乗算を含むため、計算コストが高くなる可能性があります。効率的な実装のためには、事前に計算されたルックアップテーブル（Teテーブル、Tdテーブルなど）が使用されることが一般的です。これらのテーブルは、SubBytesとMixColumnsの操作を組み合わせたもので、一度のテーブル参照とXOR演算で複数の操作を同時に実行できるように最適化されています。

## 技術的詳細

このコミットでは、AESのコアアルゴリズムがGo言語で純粋に実装されています。特に注目すべきは、FIPS 197（Advanced Encryption Standardの公式仕様）に厳密に従っている点と、パフォーマンスを考慮してルックアップテーブルを多用している点です。

### `src/lib/crypto/aes/const.go`

このファイルは、AESの数学的基盤となる定数とルックアップテーブルを定義しています。

*   `poly`: AESのGF(2^8)演算で使用される既約多項式 `x^8 + x^4 + x^3 + x + 1` を表す定数 `0x11b` (バイナリで `100011011`)。
*   `powx`: GF(2)上での `x` のべき乗を計算するためのテーブル。これはラウンド定数（Rcon）の生成にも使用されます。
*   `sbox0`: AESのSubBytes変換で使用されるS-Boxのルックアップテーブル。256バイトの配列で、各バイトが別のバイトに置換されます。
*   `sbox1`: AESのInvSubBytes（逆SubBytes）変換で使用される逆S-Boxのルックアップテーブル。`sbox0` の逆変換を行います。
*   `te` (T-tables for encryption): 暗号化のMixColumnsとSubBytesを組み合わせたルックアップテーブル。4つの256エントリの`uint32`配列で構成され、各エントリは4バイトの値を持ちます。これにより、一度のテーブル参照で複数の操作を効率的に行えます。
*   `td` (T-tables for decryption): 復号化のInvMixColumnsとInvSubBytesを組み合わせたルックアップテーブル。`te` と同様に、復号化処理を高速化します。

これらのテーブルは、AESの計算負荷の高い部分を事前に計算しておくことで、実行時のパフォーマンスを大幅に向上させます。

### `src/lib/crypto/aes/block.go`

このファイルには、AESのブロック暗号化・復号化の主要な関数と鍵拡張アルゴリズムが実装されています。

*   `encryptBlock(xk, src, dst []uint32)`: 128ビット（4つの`uint32`ワード）のブロックを暗号化する関数。
    *   `xk`: 鍵拡張によって生成されたラウンド鍵の配列。
    *   `src`: 暗号化する平文ブロック。
    *   `dst`: 暗号化された結果を格納するバイト配列。
    *   この関数は、AESの各ラウンド（SubBytes, ShiftRows, MixColumns, AddRoundKey）を `te` テーブルを使って効率的に実行します。最終ラウンドではMixColumnsは適用されません。
*   `decryptBlock(xk, src, dst []uint32)`: 128ビットのブロックを復号化する関数。
    *   `xk`: 復号化用のラウンド鍵の配列。
    *   `src`: 復号化する暗号文ブロック。
    *   `dst`: 復号化された結果（平文）を格納するバイト配列。
    *   暗号化の逆の操作（InvShiftRows, InvSubBytes, InvMixColumns, AddRoundKey）を `td` テーブルを使って実行します。
*   `subw(w uint32)`: 32ビットワードの各バイトにS-Box変換を適用するヘルパー関数。
*   `rotw(w uint32)`: 32ビットワードを左に8ビット巡回シフトするヘルパー関数。
*   `expandKey(key, enc, dec []uint32)`: 秘密鍵から暗号化用および復号化用のラウンド鍵を生成する鍵拡張アルゴリズム。
    *   `key`: 元の秘密鍵。
    *   `enc`: 暗号化用のラウンド鍵を格納する配列。
    *   `dec`: 復号化用のラウンド鍵を格納する配列。
    *   この関数は、FIPS 197の鍵拡張アルゴリズム（SubWord, RotWord, Rconの適用）を実装しています。復号化鍵は、暗号化鍵から逆MixColumns変換を適用して導出されます。

### `src/lib/crypto/aes/aes_test.go`

## コアとなるコードの解説

このコミットの核心は、AESのブロック暗号化・復号化と鍵拡張のアルゴリズムをGo言語で効率的に実装した点にあります。

### `const.go` のルックアップテーブル

`const.go` で定義されている `te` と `td` テーブルは、AESのパフォーマンス最適化において非常に重要です。AESの各ラウンドでは、バイト置換（SubBytes）と列混合（MixColumns）という2つの主要な非線形・線形変換が行われます。これらの操作は、それぞれGF(2^8)上での複雑な計算を含みます。

`te` テーブルは、`SubBytes(x) * {02} * x^3 + SubBytes(x) * {01} * x^2 + SubBytes(x) * {01} * x^1 + SubBytes(x) * {03} * x^0` のような形で、SubBytesとMixColumnsの操作を組み合わせた結果を事前に計算して格納しています。これにより、`encryptBlock` 関数内で各バイトに対して個別にSubBytesとMixColumnsを計算する代わりに、単一のテーブル参照とXOR演算でこれらの操作をまとめて実行できます。これは、特にソフトウェア実装において大きな速度向上をもたらします。`td` テーブルも同様に、復号化の逆操作を高速化します。

### `block.go` の `encryptBlock` と `decryptBlock`

`encryptBlock` 関数は、128ビットの入力ブロック（4つの`uint32`ワード）を暗号化します。
最初のラウンドでは、入力と最初のラウンド鍵をXORします。
中間ラウンドでは、`te` テーブルを駆使して、SubBytes、ShiftRows、MixColumns、AddRoundKeyの各変換を効率的に行います。各`uint32`ワードは4バイトから構成されており、`>>` や `& 0xff` を使って各バイトを抽出し、`te` テーブルの適切なエントリを参照しています。ShiftRowsは、`te` テーブルのインデックス計算と結果のワードのビットシフトによって暗黙的に実現されています。
最終ラウンドではMixColumnsは適用されず、S-Box変換とラウンド鍵のXORのみが行われます。

`decryptBlock` 関数は、暗号化の逆のプロセスを実行します。復号化では、暗号化とは異なる逆変換（InvShiftRows, InvSubBytes, InvMixColumns）が必要となります。これらの逆変換も、`td` テーブルと逆S-Box (`sbox1`) を使用して効率的に実装されています。復号化のラウンド鍵は、鍵拡張時に暗号化鍵から導出されたものが使用されます。

### `block.go` の `expandKey`

`expandKey` 関数は、AESの鍵拡張アルゴリズムを実装しています。これは、元の秘密鍵から、各ラウンドで使用されるラウンド鍵を生成するプロセスです。
鍵拡張は、以下のステップを繰り返します。
1.  前のワードを左に8ビット巡回シフト (`rotw`)。
2.  各バイトにS-Box変換を適用 (`subw`)。
3.  特定のラウンドでは、ラウンド定数 (`powx` から導出) をXORする。
4.  鍵長が256ビットの場合、特定のラウンドで追加のS-Box変換を適用する。
5.  前のラウンド鍵ワードとXORして新しいラウンド鍵ワードを生成する。

復号化用の鍵は、暗号化用の鍵から逆MixColumns変換を適用して導出されます。これにより、復号化処理も効率的に行えるようになります。

### `aes_test.go` のテスト戦略

`aes_test.go` は、実装の正確性を保証するための重要な役割を担っています。
*   **低レベルな数学的演算の検証**: `TestPowx`, `TestMul`, `TestSboxes` は、GF(2^8)上での基本的な演算やS-Boxの性質（S-Boxと逆S-Boxが互いに逆変換であること）を検証します。
*   **ルックアップテーブルの検証**: `TestTe`, `TestTd` は、`const.go` で定義された `te` および `td` テーブルが、AESの仕様に従って正しく計算されていることを検証します。これらのテストは、テーブルを生成するコードとしても機能し得るとコメントされています。
*   **FIPS 197テストベクトルによる検証**: 最も重要なのは、FIPS 197の公式ドキュメントに記載されているテストベクトル (`keyTests`, `encryptTests`) を用いて、鍵拡張、ブロック暗号化、ブロック復号化の各機能が正しく動作することを確認している点です。これにより、実装が標準規格に準拠していることが保証されます。

これらのテストは、暗号ライブラリの信頼性を確立するために不可欠であり、実装が仕様通りに機能することを証明する強力な証拠となります。

+// Multiply b and c as GF(2) polynomials modulo poly
+func mul(b, c uint32) uint32 {
+	i := b;
+	j := c;
+	s := uint32(0);
+	for k := uint32(1); k < 0x100 && j != 0; k <<= 1 {
+		// Invariant: k == 1<<n, i == b * xⁿ
+
+		if j & k != 0 {
+			// s += i in GF(2); xor in binary
+			s ^= i;
+			j ^= k;	// turn off bit to end loop early
+		}
+
+		// i *= x in GF(2) modulo the polynomial
+		i <<= 1;
+		if i & 0x100 != 0 {
+			i ^= poly;
+		}
+	}
+	return s;
+}

+// Encrypt one block from src into dst, using the expanded key xk.
+func encryptBlock(xk, src, dst []uint32) {
+	var s0, s1, s2, s3, t0, t1, t2, t3 uint32;
+
+	// First round just XORs input with key.
+	s0 = src[0] ^ xk[0];
+	s1 = src[1] ^ xk[1];
+	s2 = src[2] ^ xk[2];
+	s3 = src[3] ^ xk[3];
+
+	// Middle rounds shuffle using tables.
+	// Number of rounds is set by length of expanded key.
+	nr := len(xk)/4 - 2;	// - 2: one above, one more below
+	k := 4;\
+	for r := 0; r < nr; r++ {\
+		t0 = xk[k+0] ^ te[0][s0>>24] ^ te[1][s1>>16 & 0xff] ^ te[2][s2>>8 & 0xff] ^ te[3][s3 & 0xff];\
+		t1 = xk[k+1] ^ te[0][s1>>24] ^ te[1][s2>>16 & 0xff] ^ te[2][s3>>8 & 0xff] ^ te[3][s0 & 0xff];\
+		t2 = xk[k+2] ^ te[0][s2>>24] ^ te[1][s3>>16 & 0xff] ^ te[2][s0>>8 & 0xff] ^ te[3][s1 & 0xff];\
+		t3 = xk[k+3] ^ te[0][s3>>24] ^ te[1][s0>>16 & 0xff] ^ te[2][s1>>8 & 0xff] ^ te[3][s2 & 0xff];\
+		k += 4;\
+		s0, s1, s2, s3 = t0, t1, t2, t3;\
+	}\
+
+	// Last round uses s-box directly and XORs to produce output.
+	s0 = uint32(sbox0[t0>>24])<<24 | uint32(sbox0[t1>>16 & 0xff])<<16 | uint32(sbox0[t2>>8 & 0xff])<<8 | uint32(sbox0[t3 & 0xff]);\
+	s1 = uint32(sbox0[t1>>24])<<24 | uint32(sbox0[t2>>16 & 0xff])<<16 | uint32(sbox0[t3>>8 & 0xff])<<8 | uint32(sbox0[t0 & 0xff]);\
+	s2 = uint32(sbox0[t2>>24])<<24 | uint32(sbox0[t3>>16 & 0xff])<<16 | uint32(sbox0[t0>>8 & 0xff])<<8 | uint32(sbox0[t1 & 0xff]);\
+	s3 = uint32(sbox0[t3>>24])<<24 | uint32(sbox0[t0>>16 & 0xff])<<16 | uint32(sbox0[t1>>8 & 0xff])<<8 | uint32(sbox0[t2 & 0xff]);\
+
+	dst[0] = s0 ^ xk[k+0];\
+	dst[1] = s1 ^ xk[k+1];\
+	dst[2] = s2 ^ xk[k+2];\
+	dst[3] = s3 ^ xk[k+3];
+}\

+// Decrypt one block from src into dst, using the expanded key xk.
+func decryptBlock(xk, src, dst []uint32) {
+	var s0, s1, s2, s3, t0, t1, t2, t3 uint32;
+
+	// First round just XORs input with key.
+	s0 = src[0] ^ xk[0];
+	s1 = src[1] ^ xk[1];
+	s2 = src[2] ^ xk[2];
+	s3 = src[3] ^ xk[3];
+
+	// Middle rounds shuffle using tables.
+	// Number of rounds is set by length of expanded key.
+	nr := len(xk)/4 - 2;	// - 2: one above, one more below
+	k := 4;\
+	for r := 0; r < nr; r++ {\
+		t0 = xk[k+0] ^ td[0][s0>>24] ^ td[1][s3>>16 & 0xff] ^ td[2][s2>>8 & 0xff] ^ td[3][s1 & 0xff];\
+		t1 = xk[k+1] ^ td[0][s1>>24] ^ td[1][s0>>16 & 0xff] ^ td[2][s3>>8 & 0xff] ^ td[3][s2 & 0xff];\
+		t2 = xk[k+2] ^ td[0][s2>>24] ^ td[1][s1>>16 & 0xff] ^ td[2][s0>>8 & 0xff] ^ td[3][s3 & 0xff];\
+		t3 = xk[k+3] ^ td[0][s3>>24] ^ td[1][s2>>16 & 0xff] ^ td[2][s1>>8 & 0xff] ^ td[3][s0 & 0xff];\
+		k += 4;\
+		s0, s1, s2, s3 = t0, t1, t2, t3;\
+	}\
+
+	// Last round uses s-box directly and XORs to produce output.
+	s0 = uint32(sbox1[t0>>24])<<24 | uint32(sbox1[t3>>16 & 0xff])<<16 | uint32(sbox1[t2>>8 & 0xff])<<8 | uint32(sbox1[t1 & 0xff]);\
+	s1 = uint32(sbox1[t1>>24])<<24 | uint32(sbox1[t0>>16 & 0xff])<<16 | uint32(sbox1[t3>>8 & 0xff])<<8 | uint32(sbox1[t2 & 0xff]);\
+	s2 = uint32(sbox1[t2>>24])<<24 | uint32(sbox1[t1>>16 & 0xff])<<16 | uint32(sbox1[t0>>8 & 0xff])<<8 | uint32(sbox1[t3 & 0xff]);\
+	s3 = uint32(sbox1[t3>>24])<<24 | uint32(sbox1[t2>>16 & 0xff])<<16 | uint32(sbox1[t1>>8 & 0xff])<<8 | uint32(sbox1[t0 & 0xff]);\
+
+	dst[0] = s0 ^ xk[k+0];\
+	dst[1] = s1 ^ xk[k+1];\
+	dst[2] = s2 ^ xk[k+2];\
+	dst[3] = s3 ^ xk[k+3];
+}\

+// Apply sbox0 to each byte in w.\
+func subw(w uint32) uint32 {\
+	return\
+		uint32(sbox0[w>>24])<<24 |\
+		uint32(sbox0[w>>16 & 0xff])<<16 |\
+		uint32(sbox0[w>>8 & 0xff])<<8 |\
+		uint32(sbox0[w & 0xff]);\
+}\
+
+// Rotate\
+func rotw(w uint32) uint32 {\
+	return w<<8 | w>>24;\
+}\

+// Key expansion algorithm.  See FIPS-197, Figure 11.\
+// Their rcon[i] is our powx[i-1] << 24.\
+func expandKey(key, enc, dec []uint32) {\
+	// Encryption key setup.\
+	var i int;\
+	nk := len(key);\
+	for i = 0; i < nk; i++ {\
+		enc[i] = key[i];\
+	}\
+	for ; i < len(enc); i++ {\
+		t := enc[i-1];\
+		if i % nk == 0 {\
+			t = subw(rotw(t)) ^ (uint32(powx[i/nk - 1]) << 24);\
+		} else if nk > 6 && i % nk == 4 {\
+			t = subw(t);\
+		}\
+		enc[i] = enc[i-nk] ^ t;\
+	}\
+
+	// Derive decryption key from encryption key.\
+	// Reverse the 4-word round key sets from enc to produce dec.\
+	// All sets but the first and last get the MixColumn transform applied.\
+	if dec == nil {\
+		return;\
+	}\
+	n := len(enc);\
+	for i := 0; i < n; i += 4 {\
+		ei := n - i - 4;\
+		for j := 0; j < 4; j++ {\
+			x := enc[ei+j];\
+			if i > 0 && i+4 < n {\
+				x = td[0][sbox0[x>>24]] ^ td[1][sbox0[x>>16 & 0xff]] ^ td[2][sbox0[x>>8 & 0xff]] ^ td[3][sbox0[x & 0xff]];\
+			}\
+			dec[i+j] = x;\
+		}\
+	}\
+}\
