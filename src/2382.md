+// Negative count means read until EOF.
+func readdirnames(file *File, count int) (names []string, err Error) {
+	// If this file has no dirinfo, create one.
+	if file.dirinfo == nil {
+		file.dirinfo = new(dirInfo);
+		// The buffer must be at least a block long.
+		// TODO(r): use fstatfs to find fs block size.
+		file.dirinfo.buf = make([]byte, blockSize);
+	}
+	d := file.dirinfo;
+	size := count;
+	if size < 0 {
+		size = 100
+	}
+	names = make([]string, 0, size);	// Empty with room to grow.
+	for count != 0 {
+		// Refill the buffer if necessary
+		if d.bufp >= d.nbuf {
+			var errno int;
+			d.bufp = 0;
+			// Final argument is (basep *uintptr) and the syscall doesn't take nil.
+			d.nbuf, errno = syscall.Getdirentries(file.fd, d.buf, new(uintptr));
+			if errno != 0 {
+				d.nbuf = 0;
+				return names, ErrnoToError(errno)
+			}
+			if d.nbuf == 0 {
+				break	// EOF
+			}
+		}
+		// Drain the buffer
+		for count != 0 && d.bufp < d.nbuf {
+			dirent := (*syscall.Dirent)(unsafe.Pointer(&d.buf[d.bufp]));
+			if dirent.Reclen == 0 {
+				d.bufp = d.nbuf;
+				break
+			}
+			d.bufp += int(dirent.Reclen);
+			if dirent.Ino == 0 {	// File absent in directory.
+				continue
+			}
+			bytes := (*[len(dirent.Name)]byte)(unsafe.Pointer(&dirent.Name[0]));
+			var name = string(bytes[0:dirent.Namlen]);
+			if name == "." || name == ".." {	// Useless names
+				continue
+			}
+			count--;
+			if len(names) == cap(names) {
+				nnames := make([]string, len(names), 2*len(names));
+				for i := 0; i < len(names); i++ {
+					nnames[i] = names[i]
+				}
+				names = nnames;
+			}
+			names = names[0:len(names)+1];
+			names[len(names)-1] = name;
+		}
+	}
+	return names, nil
+}

+package os
+
+import syscall "syscall"
+import os "os"
+
+func isSymlink(stat *syscall.Stat_t) bool {
+	return stat.Mode & syscall.S_IFMT == syscall.S_IFLNK
+}
+
+func dirFromStat(name string, dir *Dir, lstat, stat *syscall.Stat_t) *Dir {
+	dir.Dev = uint64(stat.Dev);
+	dir.Ino = stat.Ino;
+	dir.Nlink = uint64(stat.Nlink);
+	dir.Mode = uint32(stat.Mode);
+	dir.Uid = stat.Uid;
+	dir.Gid = stat.Gid;
+	dir.Rdev = uint64(stat.Rdev);
+	dir.Size = uint64(stat.Size);
+	dir.Blksize = uint64(stat.Blksize);
+	dir.Blocks = uint64(stat.Blocks);
+	dir.Atime_ns = uint64(syscall.TimespecToNsec(stat.Atimespec));
+	dir.Mtime_ns = uint64(syscall.TimespecToNsec(stat.Mtimespec));
+	dir.Ctime_ns = uint64(syscall.TimespecToNsec(stat.Ctimespec));
+	for i := len(name) - 1; i >= 0; i-- {
+		if name[i] == '/' {
+			name = name[i+1:len(name)];
+			break;
+		}
+	}
+	dir.Name = name;
+	if isSymlink(lstat) && !isSymlink(stat) {
+		dir.FollowedSymlink = true;
+	}
+	return dir;

## 変更の背景
このコミットは、Go言語の初期開発段階において、Darwin (macOS) オペレーティングシステム上の386アーキテクチャでファイルシステム操作を適切にサポートするために必要なファイルが不足していた問題に対処するものです。コミットメッセージにある「g4 nothave」という記述は、当時のGoのビルドシステムや開発環境において、これらのDarwin固有のファイルがまだ含まれていなかった、あるいは認識されていなかった状況を示唆しています。

Go言語は設計当初からクロスプラットフォーム対応を重視していましたが、各OSやアーキテクチャに特化した低レベルなシステムコールへのバインディングや、それらを抽象化する標準ライブラリの実装は段階的に進められていました。このコミットは、`os`パッケージがDarwin 386環境でディレクトリの内容を読み取ったり、ファイルのメタデータ（stat情報）を取得したりする機能を確立するための重要なステップでした。これにより、GoプログラムがmacOS 386環境でファイルシステムを操作する際の基本的な機能が提供されることになります。

## 前提知識の解説

### Go言語の`os`パッケージ
`os`パッケージは、オペレーティングシステムが提供する機能（ファイルシステム操作、プロセス管理、環境変数など）へのプラットフォーム非依存なインターフェースを提供します。しかし、その内部では各OS固有のシステムコールを呼び出す必要があります。

### `syscall`パッケージ
`syscall`パッケージは、Goプログラムから低レベルなシステムコールを直接呼び出すためのインターフェースを提供します。OS固有の定数、構造体、関数などが定義されており、`os`パッケージのような高レベルな抽象化レイヤーの基盤となります。

### Darwin (macOS)
Appleが開発したUnix系オペレーティングシステムです。Go言語は、macOSを含む様々なOSをサポートしています。

### 386アーキテクチャ
Intel 80386プロセッサに由来する32ビットのx86命令セットアーキテクチャを指します。Go言語の初期には、32ビットシステムも主要なターゲットの一つでした。

### `Dirent`構造体
Unix系システムにおいて、ディレクトリ内のエントリ（ファイルやサブディレクトリ）の情報を表す構造体です。`readdir`や`getdirentries`といったシステムコールによって返されるデータの一部として使用されます。通常、エントリのinode番号、レコード長、ファイルタイプ、ファイル名などが含まれます。

### `Stat_t`構造体
Unix系システムにおいて、ファイルやディレクトリのメタデータ（統計情報）を表す構造体です。`stat`、`fstat`、`lstat`といったシステムコールによって返されます。ファイルのサイズ、パーミッション、所有者、最終アクセス/変更時刻、inode番号、デバイスIDなどが含まれます。

### `unsafe.Pointer`
Go言語の`unsafe`パッケージに含まれる型で、任意の型のポインタを保持できます。C言語のポインタのように型安全性を無視してメモリを操作するために使用され、システムコールとのインターフェースなど、低レベルな処理で必要となることがあります。

## 技術的詳細

### `src/lib/os/dir_darwin_386.go`
このファイルは、Darwin 386環境におけるディレクトリの内容を読み取るための`readdirnames`関数を実装しています。

*   **`readdirnames`関数**:
    *   `*File`型の引数`file`と、読み取るエントリ数を示す`count`を受け取ります。`count`が負の場合、EOFまで読み取ります。
    *   `file.dirinfo`という内部構造体を利用して、ディレクトリ読み取りの状態（バッファ、バッファ内の現在位置など）を管理します。
    *   `syscall.Getdirentries`システムコールを呼び出して、ディレクトリの内容をバイトスライス`d.buf`に読み込みます。このシステムコールは、指定されたファイルディスクリプタからディレクトリのエントリを読み取り、`Dirent`構造体の配列としてバッファに格納します。
    *   読み込んだバッファから`syscall.Dirent`構造体を`unsafe.Pointer`を使って抽出し、各エントリの情報を解析します。
    *   `dirent.Reclen`はディレクトリエントリのレコード長を示し、`dirent.Ino`はinode番号を示します。inode番号が0のエントリは、ファイルシステム上で存在しない（削除された）エントリとしてスキップされます。
    *   `dirent.Name`フィールドからファイル名を抽出し、`.`や`..`といった特殊なエントリは除外します。
    *   抽出したファイル名を`names`スライスに追加していきます。スライスの容量が不足した場合は、新しいスライスを作成して要素をコピーし、容量を倍増させることで効率的に拡張します。
    *   エラーが発生した場合は`ErrnoToError`でGoのエラー型に変換して返します。

### `src/lib/os/stat_darwin_386.go`
このファイルは、Darwin 386環境におけるファイルのメタデータ（stat情報）を処理するためのヘルパー関数を実装しています。

*   **`isSymlink`関数**:
    *   `*syscall.Stat_t`型の引数`stat`を受け取り、ファイルのモード（`stat.Mode`）がシンボリックリンク（`syscall.S_IFLNK`）であるかどうかを判定します。
    *   ファイルのモードはビットマスクとして様々なファイルタイプ情報を含んでおり、`& syscall.S_IFMT`でファイルタイプ部分を抽出し、`syscall.S_IFLNK`と比較することでシンボリックリンクかどうかを判断します。

*   **`dirFromStat`関数**:
    *   ファイル名`name`、`*Dir`型の`dir`、そして`lstat`と`stat`の2つの`*syscall.Stat_t`構造体を受け取ります。`lstat`はシンボリックリンク自体に関する情報、`stat`はシンボリックリンクが指すファイルに関する情報（シンボリックリンクでない場合は両方同じ）を保持します。
    *   `syscall.Stat_t`構造体から、デバイスID (`Dev`)、inode番号 (`Ino`)、リンク数 (`Nlink`)、モード (`Mode`)、ユーザーID (`Uid`)、グループID (`Gid`)、デバイスタイプ (`Rdev`)、サイズ (`Size`)、ブロックサイズ (`Blksize`)、ブロック数 (`Blocks`) などの情報を抽出し、`*Dir`構造体の対応するフィールドに設定します。
    *   タイムスタンプ（アクセス時刻`Atime_ns`、変更時刻`Mtime_ns`、作成時刻`Ctime_ns`）は、`syscall.TimespecToNsec`関数を使って`Timespec`構造体からナノ秒単位のUnixタイムスタンプに変換されます。
    *   ファイル名`name`は、パスの最後の要素（ファイル名自体）のみを抽出するように処理されます。
    *   `isSymlink`関数を使って、`lstat`がシンボリックリンクであり、かつ`stat`がシンボリックリンクでない場合（つまり、シンボリックリンクを辿った結果が通常のファイルである場合）、`dir.FollowedSymlink`を`true`に設定します。これは、`os`パッケージがシンボリックリンクをどのように扱うかに関する重要な情報を提供します。

## コアとなるコードの解説

### `readdirnames` (src/lib/os/dir_darwin_386.go)
この関数は、Goの`os`パッケージがディレクトリの内容をリストアップする際の、Darwin 386アーキテクチャ固有の低レベル実装です。
1.  **`dirInfo`の初期化**: `file.dirinfo`が`nil`の場合、`dirInfo`構造体を新しく作成し、`blockSize`（4096バイト）のバッファを割り当てます。これは、ディレクトリの内容をシステムコールから一度に読み込むためのメモリ領域です。
2.  **バッファのリフィル**: `d.bufp >= d.nbuf`（バッファ内の現在位置が読み込んだバイト数を超えた）場合、バッファが空になったと判断し、`syscall.Getdirentries`を呼び出してディレクトリから新しいエントリの塊を読み込みます。`file.fd`は開いているディレクトリのファイルディスクリプタです。
3.  **エントリの解析**: 読み込んだバッファ`d.buf`から、`unsafe.Pointer`を使って`syscall.Dirent`構造体を一つずつ取り出します。これは、C言語の構造体がメモリ上にどのように配置されているかをGo側で解釈する典型的なパターンです。
4.  **フィルタリング**: `dirent.Ino == 0`（inode番号が0）のエントリや、`dirent.Name`が`.`または`..`であるエントリは、Goの`os`パッケージのユーザーには不要な情報であるためスキップされます。
5.  **名前の抽出と追加**: `dirent.Name`から実際のファイル名を抽出し、結果のスライス`names`に追加します。`len(names) == cap(names)`の場合、スライスを倍の容量に拡張して効率的な追加を可能にします。

### `dirFromStat` (src/lib/os/stat_darwin_386.go)
この関数は、Darwin 386環境で`stat`システムコールから得られた低レベルなファイル情報を、Goの`os`パッケージが提供する高レベルな`Dir`構造体に変換する役割を担います。
1.  **メタデータのマッピング**: `syscall.Stat_t`構造体の各フィールド（`Dev`, `Ino`, `Nlink`, `Mode`, `Uid`, `Gid`, `Rdev`, `Size`, `Blksize`, `Blocks`）が、`os.Dir`構造体の対応するフィールドに直接マッピングされます。これにより、OS固有の情報をGoの抽象化されたファイル情報として利用できるようになります。
2.  **タイムスタンプの変換**: `stat.Atimespec`, `stat.Mtimespec`, `stat.Ctimespec`は`syscall.Timespec`型であり、これを`syscall.TimespecToNsec`関数を使ってナノ秒単位の整数に変換し、`os.Dir`のタイムスタンプフィールドに設定します。
3.  **ファイル名の整形**: `name`引数からパスの最後の要素（ファイル名）のみを抽出します。これは、`stat`情報が完全なパスではなく、ファイル名のみを必要とする場合に有用です。
4.  **シンボリックリンクの処理**: `isSymlink`関数を用いて、`lstat`（シンボリックリンク自体への`stat`）と`stat`（シンボリックリンクが指す実体への`stat`）を比較し、シンボリックリンクが辿られた結果であるかどうかを`dir.FollowedSymlink`フラグで示します。

これらの実装は、Go言語が異なるOSのファイルシステムAPIの差異を吸収し、統一された`os`パッケージのインターフェースを提供する上で不可欠な部分です。
