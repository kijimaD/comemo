+// ----------------------------------------------------------------------------
+// Parsing
+
+// A single error in the parsed file.
+type parseError struct {
+	src []byte;	// source before error
+	line int;	// line number of error
+	msg string;	// error message
+}
+
+
+// All the errors in the parsed file, plus surrounding source code.
+// Each error has a slice giving the source text preceding it
+// (starting where the last error occurred).  The final element in list[]
+// has msg = \"\", to give the remainder of the source code.
+// This data structure is handed to the templates parseerror.txt and parseerror.html.
+//
+type parseErrors struct {
+	filename string;	// path to file
+	list []parseError;	// the errors
+	src []byte;	// the file\'s entire source code
+}
+
+
+// Parses a file (path) and returns the corresponding AST and
+// a sorted list (by file position) of errors, if any.
+//
+func parse(path string, mode uint) (*ast.Program, *parseErrors) {
+	src, err := io.ReadFile(path);
+	if err != nil {
+		log.Stderrf(\"ReadFile %s: %v\", path, err);
+		errs := []parseError{parseError{nil, 0, err.String()}};
+		return nil, &parseErrors{path, errs, nil};
+	}
+
+	prog, err := parser.Parse(src, mode);
+	if err != nil {
+		// sort and convert error list
+		if errors, ok := err.(parser.ErrorList); ok {
+			sort.Sort(errors);
+			errs := make([]parseError, len(errors) + 1);	// +1 for final fragment of source
+			offs := 0;
+			for i, r := range errors {
+				// Should always be true, but check for robustness.
+				if 0 <= r.Pos.Offset && r.Pos.Offset <= len(src) {
+					errs[i].src = src[offs : r.Pos.Offset];
+					offs = r.Pos.Offset;
+				}
+				errs[i].line = r.Pos.Line;
+				errs[i].msg = r.Msg;
+			}
+			errs[len(errors)].src = src[offs : len(src)];
+			return nil, &parseErrors{path, errs, src};
+		} else {
+			// TODO should have some default handling here to be more robust
+			panic(\"unreachable\");
+		}
+	}
+
+	return prog, nil;
+}

+// Convert x, whatever it is, to text form.
+func toText(x interface{}) []byte {
+	type String interface { String() string }
+
+	switch v := x.(type) {
+	case []byte:
+		return v;
+	case string:
+		return io.StringBytes(v);
+	case String:
+		return io.StringBytes(v.String());
+	case ast.Decl:
+		return nodeText(v, printer.ExportsOnly);
+	case ast.Expr:
+		return nodeText(v, printer.ExportsOnly);
+	}
+	var buf io.ByteBuffer;
+	fmt.Fprint(&buf, x);
+	return buf.Data();
+}

+func readTemplate(name string) *template.Template {
+	path := pathutil.Join(*tmplroot, name);
+	data, err := io.ReadFile(path);
+	if err != nil {
+		log.Exitf(\"ReadFile %s: %v\", path, err);
+	}

+	if err1 != nil {
+		log.Exitf(\"%s: %v\", name, err);

+func isPackageFile(dirname, filename, pakname string) bool {
+	// ignore test files
+	if strings.HasSuffix(filename, \"_test.go\") {
+		return false;
+	}
+
+	// determine package name
+	prog, errors := parse(dirname + \"/\" + filename, parser.PackageClauseOnly);
+	if prog == nil {
+		return false;
+	}
+
+	return prog != nil && prog.Name.Value == pakname;
+}
+

+// Returns the canonical URL path, the package denoted by path, and
+// the list of sub-directories in the corresponding package directory.
+// If there is no such package, the package descriptor pd is nil.
+// If there are no sub-directories, the dirs list is nil.
+func findPackage(path string) (canonical string, pd *pakDesc, dirs dirList) {
+	canonical = pathutil.Clean(Pkg + path) + \"/\";
+
+	// get directory contents, if possible
+	importpath := pathutil.Clean(path);  // no trailing \'/\'
+	dirname := pathutil.Join(*pkgroot, importpath);
+	if !isDir(dirname) {
+		return;
+	}
+
+	fd, err1 := os.Open(dirname, os.O_RDONLY, 0);
+	if err1 != nil {
+		log.Stderrf(\"open %s: %v\", dirname, err1);
+		return;
+	}
+
+	list, err2 := fd.Readdir(-1);
+	if err2 != nil {
+		log.Stderrf(\"readdir %s: %v\", dirname, err2);
+		return;
+	}\
+
+	// the package name is is the directory name within its parent
+	_, pakname := pathutil.Split(dirname);
+
+	// collect all files belonging to the package and count the
+	// number of sub-directories
+	filenames := make(map[string]bool);
+	nsub := 0;
+	for i, entry := range list {
+		switch {
+		case isGoFile(&entry) && isPackageFile(dirname, entry.Name, pakname):
+			// add file to package desc
+			if tmp, found := filenames[entry.Name]; found {
+				panic(\"internal error: same file added more than once: \" + entry.Name);
+			}
+			filenames[entry.Name] = true;
+		case isPkgDir(&entry):
+			nsub++;
+		}
+	}
+
+	// make the list of sub-directories, if any
+	var subdirs dirList;
+	if nsub > 0 {
+		subdirs = make(dirList, nsub);
+		nsub = 0;
+		for i, entry := range list {
+			if isPkgDir(&entry) {
+				// make a copy here so sorting (and other code) doesn\'t
+				// have to make one every time an entry is moved
+				copy := new(os.Dir);
+				*copy = entry;
+				subdirs[nsub] = copy;
+				nsub++;
+			}
+		}
+		sort.Sort(subdirs);
+	}
+
+	// if there are no package files, then there is no package
+	if len(filenames) == 0 {
+		return canonical, nil, subdirs;
+	}
+
+	return canonical, &pakDesc{dirname, pakname, importpath, filenames}, subdirs;
+}

+func (p *pakDesc) Doc() (*doc.PackageDoc, *parseErrors) {
+	if p == nil {
+		return nil, nil;
+	}
+
+	// compute documentation
+	var r doc.DocReader;
+	i := 0;
+	for filename := range p.filenames {
+		prog, err := parse(p.dirname + \"/\" + filename, parser.ParseComments);
+		if err != nil {
+			return nil, err;
+		}
+		if i == 0 {
+			// first file - initialize doc
+			r.Init(prog.Name.Value, p.importpath);
+		}
+		i++;
+		r.AddProgram(prog);
+	}
+
+	return r.Doc(), nil;
+}

+type PageInfo struct {
+	PDoc *doc.PackageDoc;
+	Dirs dirList;

+func servePkg(c *http.Conn, r *http.Request) {
+	path := r.Url.Path;
+	path = path[len(Pkg) : len(path)];
+	canonical, desc, dirs := findPackage(path);
+
+	if r.Url.Path != canonical {
+		http.Redirect(c, canonical, http.StatusMovedPermanently);
+		return;
+	}
+
+	pdoc, errors := desc.Doc();
+	if errors != nil {
+		serveParseErrors(c, errors);
+		return;
+	}
+
+	var buf io.ByteBuffer;
+	if false {	// TODO req.Params[\"format\"] == \"text\"
+		err := packageText.Execute(PageInfo{pdoc, dirs}, &buf);
+		if err != nil {
+			log.Stderrf(\"packageText.Execute: %s\", err);
+		}
+		serveText(c, buf.Data());
+		return;
+	}
+
+	err := packageHtml.Execute(PageInfo{pdoc, dirs}, &buf);
+	if err != nil {
+		log.Stderrf(\"packageHtml.Execute: %s\", err);
+	}
+
+	if path == \"\" {
+		path = \".\";  // don\'t display an empty path
+	}
+	servePage(c, path + \" - Go package documentation\", buf.Data());
+}

+func exec(c *http.Conn, args []string) bool {
+	r, w, err := os.Pipe();
+	if err != nil {
+		log.Stderrf(\"os.Pipe(): %v\\n\", err);
+		return false;
+	}
+
+	bin := args[0];
+	fds := []*os.File{nil, w, w};
+	if *verbose {
+		log.Stderrf(\"executing %v\", args);
+	}
+	pid, err := os.ForkExec(bin, args, os.Environ(), goroot, fds);
+	defer r.Close();
+	w.Close();
+	if err != nil {
+		log.Stderrf(\"os.ForkExec(%q): %v\\n\", bin, err);
+		return false;
+	}
+
+	var buf io.ByteBuffer;
+	io.Copy(r, &buf);
+	wait, err := os.Wait(pid, 0);
+	if err != nil {
+		os.Stderr.Write(buf.Data());
+		log.Stderrf(\"os.Wait(%d, 0): %v\\n\", pid, err);
+		return false;
+	}
+	if !wait.Exited() || wait.ExitStatus() != 0 {
+		os.Stderr.Write(buf.Data());
+		log.Stderrf(\"executing %v failed (exit status = %d)\", args, wait.ExitStatus());
+		return false;
+	}
+
+	if *verbose {
+		os.Stderr.Write(buf.Data());
+	}
+	if c != nil {
+		c.SetHeader(\"content-type\", \"text/plain; charset=utf-8\");
+		c.Write(buf.Data());
+	}
+
+	return true;

## 変更の背景

このコミットは、Go言語の初期開発段階（2009年6月）に行われたもので、Goプロジェクトの構造がまだ進化している時期に当たります。`godoc` はGoのドキュメンテーション生成と表示のための重要なツールとして構想されており、その役割の重要性から、Goの公式ツールチェーンの一部として明確に位置付けられる必要がありました。

当初 `lib/godoc` に置かれていた `godoc` を `src/cmd/godoc` へ移動することで、以下の目的が達成されました。

1.  **標準コマンドとしての確立**: `src/cmd` ディレクトリは、Goの標準コマンド（例: `go build`, `go run` など）が配置される場所です。ここに `godoc` を移動することで、Go開発者にとって不可欠なツールとしての地位を確立しました。
2.  **ビルドシステムとの統合**: `src/cmd` に配置されたツールは、Goのメインのビルドプロセス (`make.bash`) によって自動的にビルド・インストールされます。これにより、`godoc` の配布と利用が容易になりました。
3.  **プロジェクト構造の整理**: `lib` ディレクトリはライブラリや共有コードを格納する場所ですが、`godoc` は単なるライブラリではなく、独立した実行可能なアプリケーションです。適切なディレクトリに配置することで、プロジェクト全体の構造がより論理的になりました。
4.  **Goの設計思想の反映**: Goは「ドキュメンテーションはコードの一部であるべき」という思想を強く持っています。`godoc` はこの思想を具現化するツールであり、その重要性からGoのコアリポジトリ内で特別な扱いを受けることになりました。

## 前提知識の解説

このコミットを理解するためには、以下の概念が役立ちます。

*   **Goのソースコードレイアウト**: Goプロジェクトのソースコードは慣習的に特定のディレクトリ構造に従います。
    *   `src/`: Goのソースコードが置かれるルートディレクトリ。
    *   `src/pkg/`: 標準ライブラリのパッケージが置かれる場所。
    *   `src/cmd/`: Goの標準コマンドラインツール（例: `go`, `gofmt`, `godoc` など）のソースコードが置かれる場所。
*   **`godoc` ツール**: `godoc` はGoのソースコードからドキュメンテーションを抽出し、整形して表示するツールです。コマンドラインツールとして、またはHTTPサーバーとして機能し、Goのパッケージ、関数、型などのドキュメンテーションを生成します。コメントを解析し、HTMLやプレーンテキスト形式で出力できます。
*   **Goのビルドシステム (`make.bash`)**: Goの初期のビルドシステムは `make.bash` というシェルスクリプトによって管理されていました。これは、Goのコンパイラ、リンカ、標準ライブラリ、および各種ツールをビルド・インストールする役割を担っていました。
*   **抽象構文木 (AST)**: `godoc` はGoのソースコードを解析し、その抽象構文木 (AST) を構築します。ASTはプログラムの構造を木構造で表現したもので、`godoc` はこのASTからドキュメンテーションコメントや宣言情報を抽出します。
*   **`go/parser` および `go/doc` パッケージ**:
    *   `go/parser`: Goのソースコードを解析し、ASTを生成するためのパッケージ。
    *   `go/doc`: ASTからドキュメンテーションを抽出・整形するためのパッケージ。`godoc` の中核をなすライブラリです。
*   **`text/template` パッケージ**: Goの標準ライブラリに含まれるテンプレートエンジンで、データ構造をHTMLやテキストなどの出力形式に変換するために使用されます。`godoc` はこれを利用してドキュメンテーションの表示形式を制御しています。

## 技術的詳細

このコミットの技術的な詳細としては、`godoc` の実装が `src/cmd/godoc/godoc.go` に集約され、それに伴い必要なテンプレートファイル (`package.html`, `package.txt`, `parseerror.html`, `parseerror.txt`) が `lib/godoc` から `src/cmd/godoc` のテンプレートルート (`tmplroot`) に移動されたことが挙げられます。

`godoc.go` のコードは、以下の主要な機能を含んでいます。

1.  **コマンドライン引数の解析**: `flag` パッケージを使用して、`http` アドレス、`goroot`、`pkgroot`、`tmplroot` などの設定を解析します。
2.  **ファイルシステム操作**: `os` パッケージを使用して、ディレクトリの存在確認、ファイルの読み込み、ディレクトリの走査などを行います。
3.  **Goソースコードの解析**: `go/parser` パッケージを使用してGoのソースファイルを解析し、ASTを生成します。エラーが発生した場合は、`parseErrors` 構造体でエラー情報を収集し、専用のテンプレートで表示します。
4.  **ドキュメンテーションの抽出と整形**: `go/doc` パッケージを使用して、解析されたASTからパッケージのドキュメンテーション (`doc.PackageDoc`) を生成します。
5.  **テンプレートによる出力**: `text/template` パッケージを使用して、抽出されたドキュメンテーションをHTMLまたはプレーンテキスト形式で整形して出力します。`fmap` という `template.FormatterMap` を定義し、`html`, `html-comment`, `text` の各フォーマットに対応する関数を登録しています。
6.  **HTTPサーバー機能**: `net/http` パッケージを使用して、`godoc` をHTTPサーバーとして起動する機能を提供します。
    *   `/` (ルート): ランディングページ (`doc/root.html`) を提供。
    *   `/doc/`: `$GOROOT/doc` からドキュメントを提供。
    *   `/src/`: `$GOROOT/src` からソースファイルを提供（Goファイルは整形して表示）。
    *   `/pkg/`: Goパッケージのドキュメンテーションを提供。
    *   `/debug/sync`: `syncCmd` が設定されている場合に、リポジトリの同期コマンドを実行するデバッグエンドポイント。
7.  **定期的な同期機能**: `syncCmd` と `syncMin` フラグが設定されている場合、指定された間隔で外部コマンド（例: `p4 sync`）を実行し、ソースコードリポジトリを最新の状態に保つゴルーチンを起動します。これは、`godoc` が常に最新のドキュメンテーションを提供できるようにするためのものです。
8.  **`Makefile` の追加**: `src/cmd/godoc/Makefile` は、`godoc` コマンドをビルドするためのルールを定義しています。これにより、Goのビルドシステムが `godoc` を自動的にコンパイルし、インストールできるようになります。
9.  **`src/cmd/make.bash` の変更**: `make.bash` スクリプトに `godoc` が追加され、Goの標準ツールとしてビルド対象に含まれるようになりました。

このコミットは、`godoc` が単なる実験的なツールから、Goエコシステムの中核をなす公式ツールへと昇格したことを明確に示しています。特に、HTTPサーバー機能と定期的な同期機能は、Goのドキュメンテーションが常に最新かつアクセスしやすい形で提供されるというGoプロジェクトの強いコミットメントを反映しています。

## コアとなるコードの変更箇所

このコミットにおけるコアとなるコードの変更箇所は以下の通りです。

1.  **`src/cmd/godoc/godoc.go` の新規追加**:
    *   `godoc` ツールのメインロジックがこのファイルに実装されました。これには、コマンドライン引数の処理、ファイルシステム操作、Goソースコードの解析、ドキュメンテーションの抽出、テンプレートによる出力、HTTPサーバー機能、および定期的なソースコード同期機能が含まれます。
2.  **`src/cmd/godoc/Makefile` の新規追加**:
    *   `godoc` バイナリをビルドするための `Makefile` が追加されました。これにより、Goのビルドシステムが `godoc` をコンパイルし、`$(HOME)/bin` にインストールする手順が定義されました。
3.  **`lib/godoc/` からのテンプレートファイルの移動**:
    *   `package.html`, `package.txt`, `parseerror.html`, `parseerror.txt` といったテンプレートファイルが、`lib/godoc` から `src/cmd/godoc` の内部で参照されるように変更されました。コミットログでは `lib/godoc/` に新規追加として表示されていますが、これは `git diff` の特性上、元の場所から削除され、新しい場所に新規追加されたように見えるためです。実質的には移動です。これらのファイルは `godoc` がドキュメンテーションを整形する際に使用する表示フォーマットを定義しています。
4.  **`src/cmd/make.bash` の変更**:
    *   `make.bash` スクリプト内の `for i in ...` ループに `godoc` が追加されました。これにより、Goのビルドプロセスにおいて `godoc` が他の標準コマンド（`cc`, `gc`, `ar` など）と同様にビルドされるようになりました。

## コアとなるコードの解説

`src/cmd/godoc/godoc.go` は、`godoc` の心臓部であり、その機能のほとんどを実装しています。

*   **`main` 関数**:
    *   コマンドライン引数を解析し、`http` フラグが指定されていればHTTPサーバーモードで、そうでなければコマンドラインモードで動作します。
    *   `GOROOT` に `chdir` した後、`readTemplates()` を呼び出して必要なテンプレートを読み込みます。
    *   HTTPサーバーモードでは、`/pkg/`、`/debug/sync`、`/` などのパスにハンドラを登録し、`http.ListenAndServe` を起動します。
    *   定期的な同期が有効な場合、`sync` ゴルーチンを起動します。
    *   コマンドラインモードでは、`findPackage` でパッケージを検索し、`desc.Doc()` でドキュメンテーションを生成し、`packageText.Execute` で標準出力に表示します。
*   **`parse` 関数**:
    *   指定されたGoソースファイルを読み込み、`go/parser.Parse` を使用してASTを生成します。
    *   構文エラーが発生した場合、`parser.ErrorList` を処理し、エラーメッセージとソースコードの断片を含む `parseErrors` 構造体を返します。
*   **`pakDesc` 構造体と `findPackage` 関数**:
    *   `pakDesc` はパッケージのディレクトリ名、パッケージ名、インポートパス、および所属するファイル名を保持します。
    *   `findPackage` は与えられたパスに対応するパッケージを `$GOROOT/src/pkg` 内で検索し、`pakDesc` とサブディレクトリのリストを返します。この関数は、パッケージのGoファイルを識別し、テストファイルを無視するロジックを含んでいます。
*   **`pakDesc.Doc()` メソッド**:
    *   `pakDesc` に含まれるGoファイル群を `parse` し、`go/doc.DocReader` を使用して `doc.PackageDoc` を構築します。これにより、パッケージ全体のドキュメンテーション情報（定数、変数、関数、型、メソッドなど）がまとめられます。
*   **テンプレート関連の関数 (`readTemplate`, `htmlFmt`, `textFmt` など)**:
    *   `readTemplate` は指定されたパスからテンプレートファイルを読み込み、`text/template.Parse` で解析します。
    *   `fmap` は、テンプレート内で使用されるフォーマッタ（例: `html`, `html-comment`）を定義し、それぞれ `template.HtmlEscape` や `doc.ToHtml` を呼び出して適切なエスケープ処理やHTML変換を行います。
*   **HTTPハンドラ (`serveFile`, `servePkg`, `serveGoSource`, `serveParseErrors`)**:
    *   これらの関数は、HTTPリクエストに応じてファイル、パッケージドキュメンテーション、Goソースコード、または解析エラーをブラウザに提供します。特に `serveGoSource` は、Goソースコードを読みやすく整形して表示する機能を提供します。
*   **`exec` および `sync` 関数**:
    *   `exec` は外部コマンドを実行し、その出力をキャプチャします。
    *   `sync` は `exec` を利用して、設定された同期コマンド（例: `p4 sync`）を定期的に実行し、`godoc` が参照するソースコードを最新の状態に保ちます。

これらのコード変更は、`godoc` がGoのソースコードを解析し、そのドキュメンテーションをコマンドラインまたはWebインターフェースを通じて提供するための包括的なフレームワークを確立したことを示しています。
